<script>
    let cart = {};

    // Stock Codes & Descriptions for sorting sequence (as per spreadsheet)
    const SEQUENCE = [
        { code: 'ER-R-AE/3600', desc: 'Aeri Rail, 40*3600mm' },
        { code: 'ER-R-AE/2400', desc: 'Aeri Rail, 40*2400mm' },
        { code: 'ER-ECAE/28-36/L50', desc: 'Universal end clamp 28-40mm Aeri' },
        { code: 'ER-IC-AE/30-40/L50', desc: 'Inter Clamp 30-40mm' },
        { code: 'WQ-TRE-3035-S', desc: 'Silver TR End Clamp' },
        { code: 'WQ-TRM-3035-S', desc: 'Silver TR Mid Clamp' },
        { code: 'WQ-TRπ-25-400', desc: 'H25 TR π MiniRail, 400mm' },
        { code: 'WQ-TRπ-60-400', desc: 'H60 TR π MiniRail, 400mm' },
        { code: 'WQ-TRU-30-120', desc: 'H30 TR U MiniRail, 120mm' },
        { code: 'EZ-GC-AE', desc: 'Grounding Clip for Aeri Rail' },
        { code: 'EZ-GL-U', desc: 'Universal Grounding Lug' },
        { code: 'WQ-TR-GL', desc: 'TR Grounding Lug' },
        { code: 'EZ-CC-PV/4', desc: 'Universal Cable Clip for PV Panels' }
    ];

    function addToCart(mode) {
        if (mode === 'rooftop') {
            const qty = parseInt(document.getElementById('rt-qty').value);
            const strings = parseInt(document.getElementById('rt-strings').value);
            const isPortrait = document.getElementById('rt-orient').value === 'portrait';
            
            // Simplified Rail Logic (Strictly 3600/2400)
            const panelWidth = isPortrait ? 1134 : 2278; // Standard sizes
            const totalRun = (panelWidth * qty) + (20 * (qty-1)) + 80;
            const rails3600 = Math.floor(totalRun / 3600);
            const remainder = totalRun % 3600;
            const rails2400 = remainder > 0 ? 1 : 0;

            updateItem('ER-R-AE/3600', rails3600 * 2 * strings);
            updateItem('ER-R-AE/2400', rails2400 * 2 * strings);
            updateItem('ER-ECAE/28-36/L50', 4 * strings);
            updateItem('ER-IC-AE/30-40/L50', (qty - 1) * 2 * strings);
            updateItem('EZ-GC-AE', (qty - 1) * 2 * strings); // Match Mid Clamps
            updateItem('EZ-GL-U', 2 * strings);
            updateItem('EZ-CC-PV/4', qty * strings);

        } else if (mode === 'shortrail') {
            const type = document.getElementById('sr-type').value;
            const qty = parseInt(document.getElementById('sr-qty').value);
            const strings = parseInt(document.getElementById('sr-strings').value);

            const railsNeeded = ((qty - 1) * 2 + 4) * strings;
            
            if (type === 'H25') updateItem('WQ-TRπ-25-400', railsNeeded);
            else if (type === 'H60') updateItem('WQ-TRπ-60-400', railsNeeded);
            else if (type === 'H30') updateItem('WQ-TRU-30-120', railsNeeded);

            updateItem('WQ-TRE-3035-S', 4 * strings);
            updateItem('WQ-TRM-3035-S', (qty - 1) * 2 * strings);
            updateItem('WQ-TR-GL', 2 * strings);
            updateItem('EZ-CC-PV/4', qty * strings);
        }

        renderBOM();
    }

    function updateItem(code, amount) {
        if (!cart[code]) cart[code] = 0;
        cart[code] += amount;
    }

    function renderBOM() {
        const body = document.getElementById('bom-body');
        body.innerHTML = '';
        
        SEQUENCE.forEach(item => {
            if (cart[item.code] && cart[item.code] > 0) {
                const row = `<tr>
                    <td>${item.code}</td>
                    <td>${item.desc}</td>
                    <td class="accent-text">${cart[item.code]}</td>
                </tr>`;
                body.innerHTML += row;
            }
        });
    }

    function clearCart() {
        cart = {};
        renderBOM();
    }
</script>
